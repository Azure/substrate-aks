package cmd

import (
	"github.com/spf13/cobra"
	"go.ate.dev/substrate/cmd/ate-setup/internal/steps"
)

// substrate-aks adds these read-only commands to a disposable pinned checkout.
func init() {
	skip := func(*cobra.Command, []string) error { return nil }
	publishCmd.AddCommand(&cobra.Command{
		Use:    "check-images REPOSITORY TAG PLUGIN OWNED_HOST DOCKER_CONFIG_DIR",
		Hidden: true, Args: cobra.ExactArgs(5),
		PersistentPreRunE: skip, PersistentPostRunE: skip,
		RunE: func(cmd *cobra.Command, args []string) error {
			return steps.CheckAzureImages(cmd.Context(), args[0], args[1], args[2], args[3], args[4], cmd.OutOrStdout())
		},
	}, &cobra.Command{
		Use:    "release-inventory REPOSITORY TAG REVISION",
		Hidden: true, Args: cobra.ExactArgs(3),
		PersistentPreRunE: skip, PersistentPostRunE: skip,
		RunE: func(cmd *cobra.Command, args []string) error {
			return steps.AzureReleaseInventory(cmd.Context(), args[0], args[1], args[2], cmd.OutOrStdout())
		},
	}, &cobra.Command{
		Use:    "microvm-assets TEMPLATE OUT_DIR ARCH",
		Hidden: true, Args: cobra.ExactArgs(3),
		PersistentPreRunE: skip, PersistentPostRunE: skip,
		RunE: func(cmd *cobra.Command, args []string) error {
			return steps.AssembleAzureMicroVMAssets(cmd.Context(), args[0], args[1], args[2])
		},
	}, &cobra.Command{
		Use:    "resolve-check-images REPOSITORY TAG",
		Hidden: true, Args: cobra.ExactArgs(2),
		PersistentPreRunE: skip, PersistentPostRunE: skip,
		RunE: func(cmd *cobra.Command, args []string) error {
			return steps.ResolveCheckImages(cmd.Context(), args[0], args[1], cmd.InOrStdin(), cmd.OutOrStdout())
		},
	}, &cobra.Command{
		Use:    "restore-worker-pools KUBECONFIG CONTEXT",
		Hidden: true, Args: cobra.ExactArgs(2),
		PersistentPreRunE: skip, PersistentPostRunE: skip,
		RunE: func(cmd *cobra.Command, args []string) error {
			return steps.RestoreWorkerPools(cmd.Context(), args[0], args[1], cmd.OutOrStdout())
		},
	}, &cobra.Command{
		Use:    "restore-namespace-resources KUBECONFIG CONTEXT",
		Hidden: true, Args: cobra.ExactArgs(2),
		PersistentPreRunE: skip, PersistentPostRunE: skip,
		RunE: func(cmd *cobra.Command, args []string) error {
			return steps.RestoreNamespaceResources(cmd.Context(), args[0], args[1], cmd.OutOrStdout(), cmd.ErrOrStderr())
		},
	}, &cobra.Command{
		Use:    "restore-inventory KUBECONFIG CONTEXT",
		Hidden: true, Args: cobra.ExactArgs(2),
		PersistentPreRunE: skip, PersistentPostRunE: skip,
		RunE: func(cmd *cobra.Command, args []string) error {
			return steps.RestoreInventory(cmd.Context(), args[0], args[1], cmd.OutOrStdout())
		},
	}, &cobra.Command{
		Use:    "restore-census KUBECONFIG CONTEXT MODE",
		Hidden: true, Args: cobra.ExactArgs(3),
		PersistentPreRunE: skip, PersistentPostRunE: skip,
		RunE: func(cmd *cobra.Command, args []string) error {
			return steps.RestoreCensus(cmd.Context(), args[0], args[1], args[2], cmd.OutOrStdout())
		},
	})
}

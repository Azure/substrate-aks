package images

import (
	"context"
	"encoding/hex"
	"fmt"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
)

func PublishPodmanImage(ctx context.Context, rootDir, dockerRepo, imageName, tag, contextPath, koDefaultPlatforms string, extraFlags []string) (string, error) {
	// Releases label their images; no other Docker build flag is supported.
	var labels []string
	for _, flag := range extraFlags {
		label, ok := strings.CutPrefix(flag, "--label=")
		if !ok || label == "" {
			return "", fmt.Errorf("Docker build flag %q is not supported by the Podman publisher", flag)
		}
		labels = append(labels, "--label", label)
	}
	platforms, err := dockerfilePlatforms(rootDir, koDefaultPlatforms)
	if err != nil {
		return "", err
	}
	if platforms != "linux/amd64" {
		return "", fmt.Errorf("the AKS Podman publisher requires linux/amd64, got %q", platforms)
	}
	authfile := os.Getenv("REGISTRY_AUTH_FILE")
	if !filepath.IsAbs(authfile) {
		return "", fmt.Errorf("Podman publishing requires an absolute isolated REGISTRY_AUTH_FILE")
	}
	repo := strings.TrimSuffix(dockerRepo, "/") + "/" + imageName
	ref := repo + ":" + tag
	run := func(args ...string) error {
		cmd := exec.CommandContext(ctx, "podman", args...)
		cmd.Dir, cmd.Stdout, cmd.Stderr = rootDir, os.Stderr, os.Stderr
		if err := cmd.Run(); err != nil {
			return fmt.Errorf("Podman %s for %s: %w", args[0], imageName, err)
		}
		return nil
	}
	build := append([]string{"build", "--authfile", authfile, "--platform", platforms}, labels...)
	if err := run(append(build, "-t", ref, contextPath)...); err != nil {
		return "", err
	}
	dir, err := os.MkdirTemp("", "substrate-aks-podman-")
	if err != nil {
		return "", err
	}
	defer os.RemoveAll(dir)
	path := filepath.Join(dir, "digest")
	if err := run("push", "--authfile", authfile, "--digestfile", path, ref, "docker://"+ref); err != nil {
		return "", err
	}
	data, err := os.ReadFile(path)
	if err != nil {
		return "", fmt.Errorf("reading pushed Podman image digest: %w", err)
	}
	digest := strings.TrimSpace(string(data))
	hash, err := hex.DecodeString(strings.TrimPrefix(digest, "sha256:"))
	if !strings.HasPrefix(digest, "sha256:") || err != nil || len(hash) != 32 {
		return "", fmt.Errorf("Podman push did not return a valid sha256 digest")
	}
	return repo + "@" + digest, nil
}

package cmd

import (
	"github.com/spf13/cobra"
	"go.ate.dev/substrate/cmd/ate-setup/internal/steps"
)

func init() {
	publishCmd.AddCommand(&cobra.Command{
		Use:  "check-images REPOSITORY TAG PLUGIN OWNED_HOST DOCKER_CONFIG_DIR",
		Args: cobra.ExactArgs(5),
		// This read-only command needs neither Kubernetes clients nor the
		// normal configuration report on its structured-output channel.
		PersistentPreRunE: func(*cobra.Command, []string) error { return nil },
		RunE: func(cmd *cobra.Command, args []string) error {
			return steps.CheckReleaseImages(cmd.Context(), args[0], args[1], args[2], args[3], args[4], cmd.OutOrStdout())
		},
	})
}
